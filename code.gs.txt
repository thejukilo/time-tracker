function getCurrentUserEmail_() {
  const email = Session.getActiveUser().getEmail().trim().toLowerCase();

  if (!email) {
    throw new Error(
      'Could not determine the signed-in Workspace email. Check the web-app deployment settings.'
    );
  }

  return email;
}

function supabaseRequest_(path, method, payload, additionalHeaders) {
  const properties = PropertiesService.getScriptProperties();
  const baseUrl = properties.getProperty('SUPABASE_URL');
  const serviceRoleKey = properties.getProperty('SUPABASE_SERVICE_ROLE_KEY');

  if (!baseUrl || !serviceRoleKey) {
    throw new Error('Supabase Script Properties are missing.');
  }

  const response = UrlFetchApp.fetch(
    `${baseUrl.replace(/\/$/, '')}/rest/v1/${path}`,
    {
      method: method || 'get',
      contentType: 'application/json',
      headers: Object.assign({
        apikey: serviceRoleKey,
        Authorization: `Bearer ${serviceRoleKey}`,
        Prefer: 'return=representation'
      }, additionalHeaders || {}),
      payload: payload ? JSON.stringify(payload) : undefined,
      muteHttpExceptions: true
    }
  );

  const status = response.getResponseCode();
  const body = response.getContentText();

  if (status < 200 || status >= 300) {
    throw new Error(`Supabase request failed (${status}): ${body}`);
  }

  return body ? JSON.parse(body) : null;
}


function parseIsoDate_(value, fieldName) {
  const match = String(value || '').match(/^(\d{4})-(\d{2})-(\d{2})$/);

  if (!match) {
    throw new Error(`${fieldName} must use YYYY-MM-DD format.`);
  }

  const date = new Date(
    Number(match[1]),
    Number(match[2]) - 1,
    Number(match[3])
  );

  if (isNaN(date.getTime())) {
    throw new Error(`${fieldName} is invalid.`);
  }

  return date;
}

function formatIsoDate_(date) {
  return Utilities.formatDate(
    date,
    Session.getScriptTimeZone(),
    'yyyy-MM-dd'
  );
}

function buildWeekdayAllocation_(startDateValue, endDateValue, totalHoursValue) {
  const startDate = parseIsoDate_(startDateValue, 'Start date');
  const endDate = parseIsoDate_(endDateValue, 'End date');
  const totalHours = Number(totalHoursValue);

  if (endDate < startDate) {
    throw new Error('End date cannot be before start date.');
  }

  if (!isFinite(totalHours) || totalHours < 0 || totalHours > 744) {
    throw new Error('Hours must be between 0 and 744.');
  }

  const totalCents = Math.round(totalHours * 100);

  if (Math.abs(totalHours * 100 - totalCents) > 0.00001) {
    throw new Error('Hours can have a maximum of two decimal places.');
  }

  const weekdays = [];
  const cursor = new Date(startDate);

  while (cursor <= endDate) {
    const dayOfWeek = cursor.getDay();

    // Monday–Friday only.
    if (dayOfWeek !== 0 && dayOfWeek !== 6) {
      weekdays.push(new Date(cursor));
    }

    cursor.setDate(cursor.getDate() + 1);
  }

  if (!weekdays.length) {
    throw new Error('The selected period has no weekdays.');
  }

  const baseCents = Math.floor(totalCents / weekdays.length);
  const remainderCents = totalCents % weekdays.length;

  return weekdays.map((date, index) => ({
    workDate: formatIsoDate_(date),
    hours: (baseCents + (index < remainderCents ? 1 : 0)) / 100
  }));
}

function createTimeEntryInSupabase_(entry) {
  const email = getCurrentUserEmail_();
  const dailyAllocation = buildWeekdayAllocation_(
    entry.startDate,
    entry.endDate,
    entry.totalHours
  );

  return supabaseRequest_('rpc/create_time_entry', 'post', {
    p_user_email: email,
    p_department_id: entry.departmentId,
    p_category_id: entry.categoryId,
    p_project_id: entry.projectId || null,
    p_instrument_id: entry.instrumentId || null,
    p_start_date: entry.startDate,
    p_end_date: entry.endDate,
    p_total_hours: Number(entry.totalHours),
    p_daily_hours: dailyAllocation.map(day => ({
      work_date: day.workDate,
      hours: day.hours
    }))
  });
}

function sourceBoolean_(value) {
  return value === true || String(value).trim().toLowerCase() === 'true';
}

function sourceIsActive_(value) {
  // Departments in the current spreadsheet do not have a status column.
  return value === undefined ||
    value === null ||
    String(value).trim() === '' ||
    String(value).trim().toLowerCase() === 'active';
}

function uniqueByName_(items) {
  const byName = new Map();

  items.forEach(item => {
    const name = String(item.name || '').trim();

    if (name) {
      byName.set(name.toLowerCase(), Object.assign({}, item, { name }));
    }
  });

  return [...byName.values()];
}

function upsertReferenceItems_(table, items) {
  if (!items.length) {
    return [];
  }

  return supabaseRequest_(
    `${table}?on_conflict=name`,
    'post',
    items,
    {
      Prefer: 'resolution=merge-duplicates,return=representation'
    }
  );
}

function migrateReferenceDataToSupabase() {
  const departmentRows = getSourceSheetRows_('departments');
  const categoryRows = getSourceSheetRows_('categories');
  const projectRows = getSourceSheetRows_('projects');
  const instrumentRows = getSourceSheetRows_('instruments');

  const departments = uniqueByName_(
    departmentRows.map(row => ({
      name: row.name || row.department || Object.values(row)[0],
      is_active: sourceIsActive_(row.status)
    }))
  );

  const categories = uniqueByName_(
    categoryRows.map(row => ({
      name: row.name,
      requires_project: sourceBoolean_(row.projectsvisible),
      requires_instrument: sourceBoolean_(row.instrumentsvisible),
      is_active: sourceIsActive_(row.status)
    }))
  );

  const projects = uniqueByName_(
    projectRows.map(row => ({
      name: row.name || Object.values(row)[0],
      is_active: sourceIsActive_(row.status)
    }))
  );

  const instruments = uniqueByName_(
    instrumentRows.map(row => ({
      name: row.name || Object.values(row)[0],
      is_active: sourceIsActive_(row.status)
    }))
  );

  upsertReferenceItems_('departments', departments);
  upsertReferenceItems_('categories', categories);
  upsertReferenceItems_('projects', projects);
  upsertReferenceItems_('instruments', instruments);

  const result = {
    departments: departments.length,
    categories: categories.length,
    projects: projects.length,
    instruments: instruments.length
  };

  Logger.log(JSON.stringify(result));
  return result;
}

function sourceDateToIso_(value, fieldName) {
  if (value instanceof Date && !isNaN(value.getTime())) {
    return formatIsoDate_(value);
  }

  const text = String(value || '').trim();

  if (/^\d{4}-\d{2}-\d{2}$/.test(text)) {
    return text;
  }

  throw new Error(`${fieldName} is not a valid date.`);
}

}

function previewTimeEntryMigration() {
  const departments = getReferenceMap_('departments', ['id', 'name']);
  const categories = getReferenceMap_(
    'categories',
    ['id', 'name', 'requires_project', 'requires_instrument']
  );
  const projects = getReferenceMap_('projects', ['id', 'name']);
  const instruments = getReferenceMap_('instruments', ['id', 'name']);

  const sourceEntries = getSourceTimeEntryRows_();
  const issues = [];
  let validEntries = 0;

  sourceEntries.forEach(entry => {
    try {
      const departmentName = String(entry.department || '').trim();
      const categoryName = String(entry.category || '').trim();
      const projectName = String(entry.project || '').trim();
      const instrumentName = String(entry.instrument || '').trim();
      const email = String(entry.email || '').trim().toLowerCase();

      const department = departments[departmentName.toLowerCase()];
      const category = categories[categoryName.toLowerCase()];
      const project = projectName
        ? projects[projectName.toLowerCase()]
        : null;
      const instrument = instrumentName
        ? instruments[instrumentName.toLowerCase()]
        : null;

      if (!email) throw new Error('Missing user email.');
      if (!department) throw new Error(`Department not found: ${departmentName}`);
      if (!category) throw new Error(`Category not found: ${categoryName}`);
      if (projectName && !project) throw new Error(`Project not found: ${projectName}`);
      if (instrumentName && !instrument) throw new Error(`Instrument not found: ${instrumentName}`);

      if (category.requires_project && !project) {
        throw new Error('This category requires a project.');
      }

      if (category.requires_instrument && !instrument) {
        throw new Error('This category requires an instrument.');
      }

      const startDate = sourceDateToIso_(entry.startdate, 'Start date');
      const endDate = sourceDateToIso_(entry.enddate, 'End date');

      buildWeekdayAllocation_(startDate, endDate, entry.hours);
      validEntries += 1;
    } catch (error) {
      if (issues.length < 50) {
        issues.push({
          spreadsheetRow: entry.spreadsheetRow,
          reason: error.message
        });
      }
    }
  });

  const result = {
    totalSourceEntries: sourceEntries.length,
    validEntries,
    invalidEntries: sourceEntries.length - validEntries,
    sampleIssues: issues
  };

  Logger.log(JSON.stringify(result));
  return result;
}

const LEGACY_MIGRATION_NEXT_ROW_PROPERTY =
  'LEGACY_MIGRATION_NEXT_ROW';

function getSourceTimeEntryBatch_(startRow, batchSize) {
  const sheet = SpreadsheetApp
    .openById('1LBZ8lMYemyqewTMa_BZr4pssFi3Y_EjcRuu-29VAS74')
    .getSheetByName('timeEntries');

  if (!sheet) {
    throw new Error('Spreadsheet sheet not found: timeEntries');
  }

  const lastRow = sheet.getLastRow();

  if (startRow > lastRow) {
    return [];
  }

  const headers = sheet
    .getRange(1, 1, 1, sheet.getLastColumn())
    .getValues()[0]
    .map(header => String(header).trim().toLowerCase());

  const rowCount = Math.min(batchSize, lastRow - startRow + 1);
  const values = sheet
    .getRange(startRow, 1, rowCount, headers.length)
    .getValues();

  return values.map((row, offset) => {
    const entry = { spreadsheetRow: startRow + offset };

    headers.forEach((header, index) => {
      entry[header] = row[index];
    });

    return entry;
  });
}

function buildLegacyMigrationPayload_(entry, references) {
  const email = String(entry.email || '').trim().toLowerCase();
  const departmentName = String(entry.department || '').trim();
  const categoryName = String(entry.category || '').trim();
  const projectName = String(entry.project || '').trim();
  const instrumentName = String(entry.instrument || '').trim();

  const department =
    references.departments[departmentName.toLowerCase()];
  const category =
    references.categories[categoryName.toLowerCase()];
  const project = projectName
    ? references.projects[projectName.toLowerCase()]
    : null;
  const instrument = instrumentName
    ? references.instruments[instrumentName.toLowerCase()]
    : null;

  if (!email) throw new Error('Missing user email.');
  if (!department) throw new Error(`Department not found: ${departmentName}`);
  if (!category) throw new Error(`Category not found: ${categoryName}`);
  if (projectName && !project) throw new Error(`Project not found: ${projectName}`);
  if (instrumentName && !instrument) throw new Error(`Instrument not found: ${instrumentName}`);

  const startDate = sourceDateToIso_(entry.startdate, 'Start date');
  const endDate = sourceDateToIso_(entry.enddate, 'End date');
  const dailyHours = buildWeekdayAllocation_(
    startDate,
    endDate,
    entry.hours
  );

  return {
    legacy_source_key: `legacy-timeEntries-row-${entry.spreadsheetRow}`,
    allocation_method: 'legacy_equal_weekdays',
    user_email: email,
    department_id: department.id,
    category_id: category.id,
    project_id: project ? project.id : null,
    instrument_id: instrument ? instrument.id : null,
    start_date: startDate,
    end_date: endDate,
    total_hours: Number(entry.hours),
    daily_hours: dailyHours.map(day => ({
      work_date: day.workDate,
      hours: day.hours
    }))
  };
}

function migrateNextLegacyTimeEntryBatch() {
  const lock = LockService.getScriptLock();

  if (!lock.tryLock(1000)) {
    throw new Error('Another migration batch is already running.');
  }

  try {
    const properties = PropertiesService.getScriptProperties();
    const startRow = Number(
      properties.getProperty(LEGACY_MIGRATION_NEXT_ROW_PROPERTY) || 2
    );
    const batchSize = 200;
    const sourceEntries = getSourceTimeEntryBatch_(startRow, batchSize);

    if (!sourceEntries.length) {
      return {
        complete: true,
        message: 'All spreadsheet rows have been processed.'
      };
    }

    const references = getLegacyReferenceMaps_();
    const payload = [];
    const skippedByReason = {};

    sourceEntries.forEach(entry => {
      try {
        payload.push(buildLegacyMigrationPayload_(entry, references));
      } catch (error) {
        const reason = error.message;
        skippedByReason[reason] = (skippedByReason[reason] || 0) + 1;
      }
    });

    const databaseResult = payload.length
      ? supabaseRequest_(
          'rpc/migrate_legacy_time_entry_batch',
          'post',
          { p_entries: payload }
        )
      : [{ inserted_count: 0, existing_count: 0 }];

    const counts = databaseResult[0] || {
      inserted_count: 0,
      existing_count: 0
    };

    const nextRow = startRow + sourceEntries.length;
    properties.setProperty(
      LEGACY_MIGRATION_NEXT_ROW_PROPERTY,
      String(nextRow)
    );

    const result = {
      complete: sourceEntries.length < batchSize,
      processedSpreadsheetRows: sourceEntries.length,
      inserted: counts.inserted_count,
      alreadyImported: counts.existing_count,
      skipped: sourceEntries.length - payload.length,
      skippedByReason,
      nextSpreadsheetRow: nextRow
    };

    Logger.log(JSON.stringify(result));
    return result;
  } finally {
    lock.releaseLock();
  }
}

function migrateLegacyEntriesForFourMinutes() {
  const startedAt = Date.now();
  const maximumDurationMs = 240000; // Four minutes.
  let totalInserted = 0;
  let totalExisting = 0;
  let totalSkipped = 0;
  let lastResult = null;

  while (Date.now() - startedAt < maximumDurationMs) {
    lastResult = migrateNextLegacyTimeEntryBatch();

    totalInserted += lastResult.inserted;
    totalExisting += lastResult.alreadyImported;
    totalSkipped += lastResult.skipped;

    if (lastResult.complete) {
      break;
    }
  }

  const result = {
    complete: Boolean(lastResult && lastResult.complete),
    insertedThisRun: totalInserted,
    alreadyImportedThisRun: totalExisting,
    skippedThisRun: totalSkipped,
    nextSpreadsheetRow: lastResult
      ? lastResult.nextSpreadsheetRow
      : null
  };

  Logger.log(JSON.stringify(result));
  return result;
}

function formatDate_(value) {
  if (!value) return '';
  if (Object.prototype.toString.call(value) === '[object Date]' && !isNaN(value)) {
    return Utilities.formatDate(value, Session.getScriptTimeZone(), 'yyyy-MM-dd');
  }
  const text = String(value).trim();
  const match = text.match(/^(\d{4}-\d{2}-\d{2})/);
  if (match) return match[1];
  const parsed = new Date(text);
  return isNaN(parsed) ? '' : Utilities.formatDate(parsed, Session.getScriptTimeZone(), 'yyyy-MM-dd');
}

function cleanText_(value, fieldName, required) {
  const text = String(value || '').trim();
  if (required && !text) throw new Error(`${fieldName} is required.`);
  if (text.length > 250) throw new Error(`${fieldName} is too long.`);
  return text;
}

  const sheet = getSheet_(ADMINS_SHEET);
  const admins = sheet.getRange(1, 1, sheet.getLastRow(), 1).getDisplayValues()
    .flat().map(value => value.trim().toLowerCase()).filter(Boolean);
  if (!admins.includes(email)) throw new Error('Administrator access is required.');
  return email;
}

function saveUserSettingsToSheet(settings) {
  const email = currentEmail_();
  const sheet = getSheet_(USER_SETTINGS_SHEET);
  const values = sheet.getDataRange().getDisplayValues();
  const columns = headerIndex_(values[0] || []);
  const emailColumn = requiredHeader_(columns, 'email');
  const projectColumn = requiredHeader_(columns, 'visibleprojects');
  const instrumentColumn = requiredHeader_(columns, 'visibleinstruments');
  const departmentColumn = requiredHeader_(columns, 'visibledepartments');
  const normaliseList = value => [...new Set((Array.isArray(value) ? value : []).map(item => String(item).trim()).filter(Boolean))].join(',');
  const settingsValues = [normaliseList(settings.visibleProjects), normaliseList(settings.visibleInstruments), normaliseList(settings.visibleDepartments)];
  const rowIndex = values.slice(1).findIndex(row => String(row[emailColumn]).trim().toLowerCase() === email);
  if (rowIndex === -1) {
    const row = new Array(values[0].length).fill('');
    row[emailColumn] = email;
    row[projectColumn] = settingsValues[0];
    row[instrumentColumn] = settingsValues[1];
    row[departmentColumn] = settingsValues[2];
    sheet.appendRow(row);
    return;
  }
  const targetRow = rowIndex + 2;
  sheet.getRange(targetRow, projectColumn + 1).setValue(settingsValues[0]);
  sheet.getRange(targetRow, instrumentColumn + 1).setValue(settingsValues[1]);
  sheet.getRange(targetRow, departmentColumn + 1).setValue(settingsValues[2]);
}

function entryFromRow_(row, rowIndex, columns) {
  return {
    rowIndex,
    startDate: formatDate_(row[columns.startdate]),
    endDate: formatDate_(row[columns.enddate]),
    department: String(row[columns.department] || ''),
    category: String(row[columns.category] || ''),
    project: String(row[columns.project] || ''),
    instrument: String(row[columns.instrument] || ''),
    hours: Number(row[columns.hours]) || 0,
    email: String(row[columns.email] || '').trim().toLowerCase()
  };
}
function validateEntry_(entry) {
  const startDate = requireDate_(entry.startDate, 'Start date');
  const endDate = requireDate_(entry.endDate, 'End date');
  if (endDate < startDate) throw new Error('End date cannot be before start date.');
  const hours = Number(entry.hours);
  if (!isFinite(hours) || hours < 0 || hours > 24 * 31) throw new Error('Hours must be between 0 and 744.');
  return {
    startDate: dateValue_(startDate), endDate: dateValue_(endDate),
    department: cleanText_(entry.department, 'Department', true),
    category: cleanText_(entry.category, 'Category', true),
    project: cleanText_(entry.project, 'Project', false),
    instrument: cleanText_(entry.instrument, 'Instrument', false), hours
  };
}

function getTimeEntriesForCurrentUser(filters) {
  const email = currentEmail_();
  const options = filters || {};
  const startDate = options.startDate ? requireDate_(options.startDate, 'From date') : '';
  const endDate = options.endDate ? requireDate_(options.endDate, 'To date') : '';
  if (startDate && endDate && endDate < startDate) throw new Error('The end date cannot be before the start date.');
  const pageSize = Math.min(Math.max(Number(options.pageSize) || 25, 1), MAX_HISTORY_PAGE_SIZE);
  const requestedPage = Math.max(Number(options.page) || 1, 1);
  const context = timeEntryContext_();
  const entries = context.values.slice(1).map((row, index) => entryFromRow_(row, index + 2, context.columns))
    .filter(entry => entry.email === email && (!startDate || entry.startDate >= startDate) && (!endDate || entry.endDate <= endDate))
    .sort((a, b) => b.startDate.localeCompare(a.startDate) || b.rowIndex - a.rowIndex);
  const total = entries.length;
  const pageCount = Math.max(Math.ceil(total / pageSize), 1);
  const page = Math.min(requestedPage, pageCount);
  return { entries: entries.slice((page - 1) * pageSize, page * pageSize), total, page, pageCount };
}
function getLatestEntriesFromSheet() { return getTimeEntriesForCurrentUser({ page: 1, pageSize: 10 }).entries; }
function assertEntryOwner_(rowIndex, context, email) {
  const sheetRow = Number(rowIndex);
  if (!Number.isInteger(sheetRow) || sheetRow < 2 || sheetRow > context.values.length) throw new Error('The selected entry no longer exists.');
  if (String(context.values[sheetRow - 1][context.columns.email] || '').trim().toLowerCase() !== email) throw new Error('You can only change your own time entries.');
  return sheetRow;
}
function updateTimeEntry(rowIndex, updatedEntry) {
  const email = currentEmail_();
  const valid = validateEntry_(updatedEntry);
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    const context = timeEntryContext_();
    const targetRow = assertEntryOwner_(rowIndex, context, email);
    const currentRow = context.values[targetRow - 1];
    currentRow[context.columns.startdate] = valid.startDate;
    currentRow[context.columns.enddate] = valid.endDate;
    currentRow[context.columns.department] = valid.department;
    currentRow[context.columns.category] = valid.category;
    currentRow[context.columns.project] = valid.project;
    currentRow[context.columns.instrument] = valid.instrument;
    currentRow[context.columns.hours] = valid.hours;
    context.sheet.getRange(targetRow, 1, 1, currentRow.length).setValues([currentRow]);
  } finally {
    lock.releaseLock();
  }
  return 'updated';
}

function getLoggedHoursSummary() {
  const email = currentEmail_();
  const context = timeEntryContext_();
  const now = new Date();
  const firstOfMonth = new Date(now.getFullYear(), now.getMonth(), 1);
  const startOfWeek = new Date(now.getFullYear(), now.getMonth(), now.getDate());
  startOfWeek.setDate(startOfWeek.getDate() - ((startOfWeek.getDay() + 6) % 7));
  return context.values.slice(1).reduce((summary, row) => {
    const entry = entryFromRow_(row, 0, context.columns);
    if (entry.email !== email || !entry.startDate) return summary;
    const date = dateValue_(entry.startDate);
    if (date >= firstOfMonth) summary.monthHours += entry.hours;
    if (date >= startOfWeek) summary.weekHours += entry.hours;
    return summary;
  }, { monthHours: 0, weekHours: 0 });
}

function addItem(listName, item, isRemoval) {
  validListSheet_(listName);
  const email = currentEmail_();
  const name = cleanText_(item, 'Item name', true);
  const sheet = getSheet_(listName);
  const values = sheet.getDataRange().getValues();
  const columns = headerIndex_(values[0] || []);
  const nameColumn = requiredHeader_(columns, 'name');
  const statusColumn = requiredHeader_(columns, 'status');
  const matchingActive = values.slice(1).some(row => String(row[nameColumn]).trim().toLowerCase() === name.toLowerCase() && String(row[statusColumn]).toLowerCase() === 'active');
  const matchingDraft = values.slice(1).some(row => String(row[nameColumn]).trim().toLowerCase() === name.toLowerCase() && String(row[statusColumn]).toLowerCase() !== 'active');
  if (isRemoval && !matchingActive) throw new Error('Only active items can be proposed for removal.');
  if (!isRemoval && matchingActive) throw new Error('This item already exists.');
  if (matchingDraft) throw new Error('This item already has a pending suggestion.');
  const row = new Array(values[0].length).fill('');
  row[nameColumn] = name;
  row[statusColumn] = isRemoval ? `draft-remove:${email}` : email;
  sheet.appendRow(row);
}

function approveItem(type, name) {
  requireAdmin_(); validListSheet_(type);
  const sheet = getSheet_(type);
  const values = sheet.getDataRange().getValues();
  const columns = headerIndex_(values[0] || []);
  const nameColumn = requiredHeader_(columns, 'name');
  const statusColumn = requiredHeader_(columns, 'status');
  const target = values.slice(1).map((row, index) => ({ row, index: index + 2 })).reverse().find(item => String(item.row[nameColumn]) === String(name) && String(item.row[statusColumn]).toLowerCase() !== 'active');
  if (!target) throw new Error('Draft item not found.');
  if (String(target.row[statusColumn]).startsWith('draft-remove:')) {
    sheet.deleteRow(target.index);
    const refreshed = sheet.getDataRange().getValues();
    const activeIndex = refreshed.slice(1).map((row, index) => ({ row, index: index + 2 })).find(item => String(item.row[nameColumn]) === String(name) && String(item.row[statusColumn]).toLowerCase() === 'active');
    if (activeIndex) sheet.deleteRow(activeIndex.index);
  } else {
    sheet.getRange(target.index, statusColumn + 1).setValue('active');
  }
}

// Supabase application API. These functions are used by ui-v3.html and keep
// the browser away from database credentials and user-email authorization.
function ttSupabaseRequest_(path, method, payload) {
  const properties = PropertiesService.getScriptProperties();
  const baseUrl = properties.getProperty('SUPABASE_URL');
  const secretKey = properties.getProperty('SUPABASE_SERVICE_ROLE_KEY');
  if (!baseUrl || !secretKey) throw new Error('Supabase connection settings are missing.');
  const response = UrlFetchApp.fetch(`${baseUrl.replace(/\/$/, '')}/rest/v1/${path}`, {
    method: method || 'get',
    contentType: 'application/json',
    headers: { apikey: secretKey, Authorization: `Bearer ${secretKey}`, Prefer: 'return=representation' },
    payload: payload ? JSON.stringify(payload) : undefined,
    muteHttpExceptions: true
  });
  const status = response.getResponseCode();
  const body = response.getContentText();
  if (status < 200 || status >= 300) throw new Error(`Supabase request failed (${status}): ${body}`);
  return body ? JSON.parse(body) : null;
}

function getTimeTrackerBootstrap() {
  const email = currentEmail_();
  const now = new Date();
  const firstOfMonth = new Date(now.getFullYear(), now.getMonth(), 1);
  const startOfWeek = new Date(now.getFullYear(), now.getMonth(), now.getDate());
  startOfWeek.setDate(startOfWeek.getDate() - ((startOfWeek.getDay() + 6) % 7));
  const monthStart = ttDateKey_(firstOfMonth);
  const weekStart = ttDateKey_(startOfWeek);
  const encodedEmail = encodeURIComponent(email);
  const [roleRow] = ttSupabaseRequest_(`user_roles?email=eq.${encodedEmail}&select=role`, 'get');
  const dailyRows = ttSupabaseRequest_(`time_entry_days?user_email=eq.${encodedEmail}&work_date=gte.${monthStart}&select=work_date,hours`, 'get');
  const recentRows = ttSupabaseRequest_(`time_entries?user_email=eq.${encodedEmail}&select=id,start_date,total_hours,department_id,category_id&order=created_at.desc&limit=5`, 'get');
  const options = ttApplyDropdownPreferences_(ttActiveOptions_(), ttDropdownPreferences_(email));
  const allDepartments = ttSupabaseRequest_('departments?select=id,name', 'get');
  const allCategories = ttSupabaseRequest_('categories?select=id,name', 'get');
  const departmentNames = allDepartments.reduce((items, item) => Object.assign(items, { [item.id]: item.name }), {});
  const categoryNames = allCategories.reduce((items, item) => Object.assign(items, { [item.id]: item.name }), {});
  const summary = dailyRows.reduce((result, row) => {
    const hours = Number(row.hours) || 0;
    result.monthHours += hours;
    if (row.work_date >= weekStart) result.weekHours += hours;
    return result;
  }, { monthHours: 0, weekHours: 0 });
  summary.monthHours = Number(summary.monthHours.toFixed(2));
  summary.weekHours = Number(summary.weekHours.toFixed(2));
  return {
    email,
    role: roleRow ? roleRow.role : 'user',
    options,
    summary,
    recentEntries: recentRows.map(row => ({
      id: row.id,
      startDate: row.start_date,
      hours: Number(row.total_hours) || 0,
      departmentName: departmentNames[row.department_id] || 'Archived department',
      categoryName: categoryNames[row.category_id] || 'Archived category'
    }))
  };
}

function submitTimeEntryWithAllocation(entry) {
  const email = currentEmail_();
  const days = Array.isArray(entry && entry.dailyHours) ? entry.dailyHours : [];
  const totalHours = Number(entry && entry.hours);
  if (!entry || !entry.startDate || !entry.endDate || !entry.departmentId || !entry.categoryId) {
    throw new Error('Please complete all required fields.');
  }
  if (!isFinite(totalHours) || totalHours < 0 || totalHours > 744) throw new Error('Hours must be between 0 and 744.');
  const allocatedHours = days.reduce((sum, day) => sum + (Number(day.hours) || 0), 0);
  if (!days.length || Math.abs(allocatedHours - totalHours) > 0.001) throw new Error('Daily allocation must equal total hours.');
  return ttSupabaseRequest_('rpc/save_time_entry_with_allocation', 'post', {
    p_user_email: email,
    p_department_id: entry.departmentId,
    p_category_id: entry.categoryId,
    p_project_id: entry.projectId || null,
    p_instrument_id: entry.instrumentId || null,
    p_start_date: entry.startDate,
    p_end_date: entry.endDate,
    p_total_hours: totalHours,
    p_daily_hours: days.map(day => ({ work_date: day.workDate, hours: Number(day.hours) }))
  });
}


function updateTimeTrackerEntry(entry) {
  const email = currentEmail_();
  const days = Array.isArray(entry && entry.dailyHours)
    ? entry.dailyHours
    : [];
  const totalHours = Number(entry && entry.hours);

  if (
    !entry ||
    !entry.id ||
    !entry.startDate ||
    !entry.endDate ||
    !entry.departmentId ||
    !entry.categoryId
  ) {
    throw new Error('Please complete all required fields.');
  }

  if (!isFinite(totalHours) || totalHours < 0 || totalHours > 744) {
    throw new Error('Hours must be between 0 and 744.');
  }

  const allocatedHours = days.reduce(
    (sum, day) => sum + (Number(day.hours) || 0),
    0
  );

  if (
    !days.length ||
    Math.abs(allocatedHours - totalHours) > 0.001
  ) {
    throw new Error('Daily allocation must equal total hours.');
  }

  return ttSupabaseRequest_(
    'rpc/update_time_entry_with_allocation',
    'post',
    {
      p_entry_id: entry.id,
      p_user_email: email,
      p_department_id: entry.departmentId,
      p_category_id: entry.categoryId,
      p_project_id: entry.projectId || null,
      p_instrument_id: entry.instrumentId || null,
      p_start_date: entry.startDate,
      p_end_date: entry.endDate,
      p_total_hours: totalHours,
      p_daily_hours: days.map(day => ({
        work_date: day.workDate,
        hours: Number(day.hours)
      }))
    }
  );
}

function getTimeTrackerHistory(filters) {
  const options = filters || {};
  const email = currentEmail_();
  const page = Math.max(Number(options.page) || 1, 1);
  const pageSize = Math.min(Math.max(Number(options.pageSize) || 25, 1), 100);
  const query = [
    `user_email=eq.${encodeURIComponent(email)}`,
    'select=id,start_date,end_date,total_hours,department_id,category_id,project_id,instrument_id,created_at',
    'order=start_date.desc,created_at.desc',
    `limit=${pageSize + 1}`,
    `offset=${(page - 1) * pageSize}`
  ];
  if (options.startDate) query.push(`start_date=gte.${encodeURIComponent(options.startDate)}`);
  if (options.endDate) query.push(`end_date=lte.${encodeURIComponent(options.endDate)}`);
  if (options.categoryId) query.push(`category_id=eq.${encodeURIComponent(options.categoryId)}`);
  if (options.departmentId) query.push(`department_id=eq.${encodeURIComponent(options.departmentId)}`);
  const rows = ttSupabaseRequest_(`time_entries?${query.join('&')}`, 'get');
  const names = ttAllReferenceNames_();
  const hasMore = rows.length > pageSize;
  return {
    entries: rows.slice(0, pageSize).map(row => ({
      id: row.id,
      startDate: row.start_date,
      endDate: row.end_date,
      hours: Number(row.total_hours) || 0,
      departmentId: row.department_id,
      categoryId: row.category_id,
      projectId: row.project_id,
      instrumentId: row.instrument_id,
      departmentName: names.departments[row.department_id] || 'Archived department',
      categoryName: names.categories[row.category_id] || 'Archived category',
      projectName: names.projects[row.project_id] || '',
      instrumentName: names.instruments[row.instrument_id] || ''
    })),
    page,
    hasMore
  };
}

// Returns dashboard aggregates. Admins may request another user's email; all
// other filter values are validated and applied server-side.
function getTimeTrackerDashboard(filters) {
  const options = typeof filters === 'object' && filters ? filters : {};
  const requesterEmail = currentEmail_();
  const [roleRow] = ttSupabaseRequest_(`user_roles?email=eq.${encodeURIComponent(requesterEmail)}&select=role`, 'get');
  const requesterRole = roleRow ? roleRow.role : 'user';
  const canViewOtherUsers = ['admin', 'superadmin'].includes(requesterRole);
  const requestedEmails = (Array.isArray(options.userEmails) ? options.userEmails : [])
    .map(email => String(email || '').trim().toLowerCase()).filter(Boolean);
  const allUsers = canViewOtherUsers && options.allUsers === true;
  const selectedEmails = !canViewOtherUsers ? [requesterEmail] : (allUsers ? null : (requestedEmails.length ? requestedEmails : [requesterEmail]));
  const today = new Date();
  const defaultStart = new Date(today.getFullYear(), today.getMonth() - 1, 1);
  const defaultEnd = new Date(today.getFullYear(), today.getMonth(), 0);
  const startDate = options.startDate ? requireDate_(options.startDate, 'From date') : ttDateKey_(defaultStart);
  const endDate = options.endDate ? requireDate_(options.endDate, 'To date') : ttDateKey_(defaultEnd);
  if (endDate < startDate) throw new Error('The end date cannot be before the start date.');
  const startValue = dateValue_(startDate);
  const endValue = dateValue_(endDate);
  const rangeDays = Math.round((endValue.getTime() - startValue.getTime()) / 86400000) + 1;
  const rawRows = ttSupabaseRequest_('rpc/dashboard_resource_rows', 'post', {
    p_start_date: startDate,
    p_end_date: endDate,
    p_user_emails: selectedEmails,
    p_department_id: options.departmentId || null,
    p_category_id: options.categoryId || null,
    p_project_id: options.projectId || null,
    p_instrument_id: options.instrumentId || null
  });
  const rows = rawRows.map(row => ({
    departmentName: row.department_name,
    categoryName: row.category_name,
    userEmail: row.user_email,
    instrumentName: row.instrument_name || '',
    projectName: row.project_name || '',
    hours: Number(row.hours) || 0
  }));
  const categoryTotals = rows.reduce((result, row) => {
    result[row.categoryName] = (result[row.categoryName] || 0) + row.hours;
    return result;
  }, {});
  const categories = Object.keys(categoryTotals)
    .map(name => ({ name, hours: Number(categoryTotals[name].toFixed(2)) }))
    .sort((a, b) => b.hours - a.hours || a.name.localeCompare(b.name));
  const totalHours = rows.reduce((sum, row) => sum + row.hours, 0);
  const averagePerWeek = totalHours / Math.max(rangeDays / 7, 1);
  return { period: { startDate, endDate }, viewerRole: requesterRole, selectedUserCount: selectedEmails ? selectedEmails.length : null, allUsers, totalHours: Number(totalHours.toFixed(2)), averagePerWeek: Number(averagePerWeek.toFixed(2)), categories, rows };
}

function submitTimeTrackerSuggestion(payload) {
  const email = currentEmail_();
  const itemType = ttReferenceTable_(String(payload && payload.itemType || ''));
  const action = String(payload && payload.action || '');
  if (!['add', 'remove'].includes(action)) throw new Error('Invalid suggestion action.');
  const itemName = String(payload && payload.itemName || '').trim();
  const targetId = String(payload && payload.targetId || '').trim();
  if (!itemName || itemName.length > 160) throw new Error('Enter a valid item name.');
  if (action === 'remove' && !targetId) throw new Error('Choose an item to remove.');
  const duplicateQuery = `item_suggestions?submitted_by=eq.${encodeURIComponent(email)}&item_type=eq.${itemType}&action=eq.${action}&item_name=eq.${encodeURIComponent(itemName)}&status=eq.pending&select=id`;
  if (ttSupabaseRequest_(duplicateQuery, 'get').length) throw new Error('You already have a pending suggestion for this item.');
  return ttSupabaseRequest_('item_suggestions', 'post', { submitted_by: email, item_type: itemType, action, item_name: itemName, target_id: targetId || null, status: 'pending' });
}

function reviewTimeTrackerSuggestion(id, decision, note) {
  const context = ttRequireAdmin_();
  if (!['approved', 'rejected'].includes(decision)) throw new Error('Invalid review decision.');
  const [suggestion] = ttSupabaseRequest_(`item_suggestions?id=eq.${encodeURIComponent(id)}&status=eq.pending&select=*`, 'get');
  if (!suggestion) throw new Error('This suggestion is no longer pending.');
  if (decision === 'approved') {
    const table = ttReferenceTable_(suggestion.item_type);
    if (suggestion.action === 'remove') {
      ttSupabaseRequest_(`${table}?id=eq.${encodeURIComponent(suggestion.target_id)}`, 'patch', { is_active: false });
    } else {
      const existing = ttSupabaseRequest_(`${table}?name=eq.${encodeURIComponent(suggestion.item_name)}&is_active=eq.true&select=id`, 'get');
      if (existing.length) throw new Error('An active item with this name already exists.');
      const item = { name: suggestion.item_name, is_active: true };
      if (table === 'categories') Object.assign(item, { requires_project: false, requires_instrument: false });
      ttSupabaseRequest_(table, 'post', item);
    }
  }
  return ttSupabaseRequest_(`item_suggestions?id=eq.${encodeURIComponent(id)}`, 'patch', { status: decision, reviewed_by: context.email, reviewed_at: new Date().toISOString(), review_note: String(note || '').trim() || null });
}

function assignTimeTrackerAdmin(email) {
  ttRequireSuperadmin_();
  const target = String(email || '').trim().toLowerCase();
  if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(target)) throw new Error('Enter a valid email address.');
  const existing = ttSupabaseRequest_(`user_roles?email=eq.${encodeURIComponent(target)}&select=email`, 'get');
  return existing.length ? ttSupabaseRequest_(`user_roles?email=eq.${encodeURIComponent(target)}`, 'patch', { role: 'admin' }) : ttSupabaseRequest_('user_roles', 'post', { email: target, role: 'admin' });
}
function setTimeTrackerUserRole(email, role) {
  const context = ttRequireSuperadmin_();
  const target = String(email || '').trim().toLowerCase();
  const nextRole = String(role || '').trim().toLowerCase();
  if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(target)) throw new Error('Enter a valid email address.');
  if (!['user', 'admin', 'superadmin'].includes(nextRole)) throw new Error('Invalid role.');
  const existingRoles = ttSupabaseRequest_('user_roles?select=email,role', 'get');
  const current = existingRoles.find(item => item.email === target);
  const superadmins = existingRoles.filter(item => item.role === 'superadmin');
  if (current && current.role === 'superadmin' && nextRole !== 'superadmin' && superadmins.length === 1) {
    throw new Error('At least one superadmin must remain. Assign another superadmin first.');
  }
  if (nextRole === 'user') {
    if (current) return ttSupabaseRequest_(`user_roles?email=eq.${encodeURIComponent(target)}`, 'delete');
    return 'unchanged';
  }
  return current ? ttSupabaseRequest_(`user_roles?email=eq.${encodeURIComponent(target)}`, 'patch', { role: nextRole }) : ttSupabaseRequest_('user_roles', 'post', { email: target, role: nextRole });
}
